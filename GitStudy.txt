git config --global user.name "你的用户名"    //全局配置
git config --global user.eamil ""
git init    //创建本地空仓库
git add 文件名    //添加文件到暂存区
git commit -m " "    //将暂存区的文件提交到本地仓库
git commit --amend    //重写上次的提交信息
git log    //查看日志
git log --oneline    //查看日志（简洁）
git reset --hard 版本号    //回滚操作
git reset --hard HEAD^    //回滚上一个版本
git reset --hard HEAD~3    //回滚前3个版本
touch 文件名    //新增文件名
git status    //查看状态
rm 文件名    //删除文件
git log 文件名    //查看文件可回滚版本
git reset 版本id 文件名    //文件历史版本回滚
git reflog    //查看当前版本库的提交历史，凡是对仓库版本进行迭代的都会出现在这个里面，包括回滚版本都会出现在这个历史中
git branch    //查看本地分支
git branch -a    //查看所有分支(本地分支和远程跟踪分支)
git branch -r    //查看远程跟踪分支
git merge 分支名    //将当前分支和指定分支合并
如果你在任何分支下创建文件，没有提交到仓库，那么它在所有仓库都是可见的，比如你在分支dev中创建了一个文件，没有使用git add和git commit提交，此时你切换到master，这个文件依旧存在的，因为你创建的文件在工作目录中，你切换仓库时git只会更新跟仓库有关的文件，无关的文件依然存放在工作区。
git push origin --delete 远程分支名    //删除远程分支名
git add origin 远程仓库地址    //关联远程仓库
git push -u origin master    //推送到远程仓库
git checkout -b new-feature    //创建并切换到新本地分支
git switch -c new-feature    //创建并切换到新本地分支（Git 2.23+ 推荐语法）
git switch/checkout 分支名    //切换分支名
